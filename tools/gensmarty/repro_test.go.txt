package blackfriday

import (
	"encoding/hex"
	"fmt"
	"os"
	"testing"
)

// Scratch-only. Asks whether smartLeftAngle's out-of-range read is reachable
// through the exported API, and not merely through the unexported Process.
//
// writeDocumentHeader passes the Title to Process without escaping it when
// Smartypants is set, so a title containing '<' but no '>' reaches
// smartLeftAngle with i == len(text). Whether that reads a stray byte or
// panics outright depends on whether []byte(string) left spare capacity,
// i.e. on the length of the title relative to Go's size classes.
func TestZZReproTitle(t *testing.T) {
	f, err := os.Create("repro-fixture.txt")
	if err != nil {
		t.Fatal(err)
	}
	defer f.Close()

	titles := []string{}
	// A '<' with no '>' at every length from 1 to 40, so the size-class
	// boundaries show up rather than being guessed at.
	for n := 1; n <= 40; n++ {
		s := ""
		for len(s) < n-1 {
			s += "a"
		}
		titles = append(titles, s+"<")
	}
	titles = append(titles, "a<b", "<", "safe <tag> title", "no angle at all")

	for _, title := range titles {
		out, panicked := runTitle(title)
		fmt.Fprintf(f, "R %d %s %s %v\n", len(title),
			hex.EncodeToString([]byte(title)), hex.EncodeToString(out), panicked)
	}
}

func runTitle(title string) (out []byte, panicked bool) {
	defer func() {
		if recover() != nil {
			panicked = true
		}
	}()
	r := NewHTMLRenderer(HTMLRendererParameters{
		Flags: CompletePage | Smartypants,
		Title: title,
	})
	out = Run([]byte("hi"), WithRenderer(r))
	return out, false
}
