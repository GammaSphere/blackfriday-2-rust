package blackfriday

import (
	"encoding/hex"
	"fmt"
	"os"
	"testing"
)

// Scratch-only generator (not part of the pinned suite). Emits measured Go
// answers for the byte helpers so the Rust port can assert against them.
// Byte strings are hex-encoded: slugify operates on bytes and may produce
// output that is not valid UTF-8, so %q would be both lossy and ambiguous.
func TestZZGenFixture(t *testing.T) {
	f, err := os.Create("helpers-fixture.txt")
	if err != nil {
		t.Fatal(err)
	}
	defer f.Close()

	fmt.Fprintln(f, "# generated by tools/genhelpers against blackfriday v2.1.0")
	fmt.Fprintln(f, "# B <byte-hex> ispunct isspace ishorizontalspace isverticalspace isletter isalnum")
	fmt.Fprintln(f, "# S <input-hex> <slugify-output-hex>")
	fmt.Fprintln(f, "# I <data-hex> <indentSize> <isIndented-result>")

	for c := 0; c < 256; c++ {
		b := byte(c)
		fmt.Fprintf(f, "B %02x %v %v %v %v %v %v\n", c,
			ispunct(b), isspace(b), ishorizontalspace(b),
			isverticalspace(b), isletter(b), isalnum(b))
	}

	slugCases := []string{
		"", "a", "-", "!", "!!!", "abc", "a b c", "Hello, World!",
		"  leading", "trailing  ", "--dashes--", "a--b", "___", "a_b",
		"123", "a1b2", "...a...", "-a-", "a-", "-a", "\t\ttabs\t\t",
		"Ünïcödé", "中文字符", "a\x00b", "\xff\xfe", "mixed 123 !@# ABC",
		"-", "--", "---", "-a", "a-", " ", "  ", "\n", "0", "z", "Z",
	}
	for _, s := range slugCases {
		in := []byte(s)
		fmt.Fprintf(f, "S %s %s\n", hex.EncodeToString(in), hex.EncodeToString(slugify(in)))
	}

	idCases := []struct {
		data string
		size int
	}{
		{"", 4}, {"\t", 4}, {"\tx", 4}, {"    ", 4}, {"   ", 4},
		{"    x", 4}, {"   x", 4}, {"        ", 8}, {"       ", 8},
		{" \t  ", 4}, {"a   ", 4}, {"    ", 0}, {"", 0}, {"  ", 2},
		{"\t   ", 8}, {"     ", 4}, {"\v\v\v\v", 4},
	}
	for _, c := range idCases {
		fmt.Fprintf(f, "I %s %d %d\n", hex.EncodeToString([]byte(c.data)), c.size,
			isIndented([]byte(c.data), c.size))
	}
}
